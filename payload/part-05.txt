wnProgress
    );
    const dynamicErosionRate = baseErosionRate * dynamicMultiplier;

    erosionBudget += dynamicErosionRate * deltaMs / 1000;

    let erosionGuard = 0;
    while (erosionBudget >= 1 && detachedCount < maxDetached && erosionGuard < 3) {
      erosionBudget -= 1;
      erosionGuard++;

      // 缺口前沿由心尖向上推进。前段推进稍快，越往上曲线越平缓。
      const erosionProgress = clamp(detachedCount / Math.max(1, maxDetached), 0, 1);
      const frontierProgress = 1 - Math.pow(1 - erosionProgress, CONFIG.erosionFrontierEase);
      const minTargetY = shapeScale * lerp(8.8, 2.6, frontierProgress);
      const source = findVisibleDripSource(minTargetY, true);

      if (!source || !source.detachForDrip()) continue;

      detachedCount++;
      drips.push(new Drip(source, true));
    }
  }

  function updateDrops(step) {
    for (let i = drips.length - 1; i >= 0; i--) {
      const d = drips[i];
      if (d.update(step) || d.y > H + 20) {
        if (d.owner) {
          d.owner.x = d.x;
          d.owner.y = d.impactY;
          d.owner.settleX = d.x;
          d.owner.settleY = d.impactY;
          d.owner.settled = true;
          d.owner.vx = rand(-0.28, 0.28);
          d.owner.vy = 0;
        }

        if (d.x > centerX - floorRX * 1.08 && d.x < centerX + floorRX * 1.08) {
          const count = Math.random() < 0.45 ? 2 : 1;
          for (let n = 0; n < count; n++) splashes.push(new Splash(d.x, d.impactY, d.colorIndex));
        }
        drips.splice(i, 1);
      }
    }

    for (let i = splashes.length - 1; i >= 0; i--) {
      const s = splashes[i];
      s.update(step);
      if (s.life <= 0) splashes.splice(i, 1);
    }
  }

  function drawAmbientGlow(pinkMix, sceneAlpha) {
    // 开场灰色阶段不出现彩色光晕，真正点亮时才从黑暗里透出粉紫光。
    const strength = clamp(pinkMix, 0, 1) * sceneAlpha;
    if (strength <= 0.001) return;

    const r = Math.max(120, shapeScale * 18);
    const g = ctx.createRadialGradient(centerX, centerY, 0, centerX, centerY, r);
    g.addColorStop(0, `rgba(195,52,207,${0.035 * strength})`);
    g.addColorStop(0.55, `rgba(123,26,163,${0.010 * strength})`);
    g.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = g;
    ctx.fillRect(centerX - r, centerY - r, r * 2, r * 2);
  }

  function drawFloorGlow(pinkMix, sceneAlpha) {
    const mix = clamp(pinkMix, 0, 1);
    if (sceneAlpha <= 0.001) return;

    ctx.save();
    ctx.translate(centerX, floorY);
    ctx.scale(1, floorRY / floorRX);

    // 灰色粒子堆的冷光。
    if (mix < 0.999) {
      const gray = ctx.createRadialGradient(0, 0, floorRX * 0.08, 0, 0, floorRX * 1.08);
      gray.addColorStop(0, `rgba(125,130,143,${0.010 * sceneAlpha * (1 - mix)})`);
      gray.addColorStop(0.52, `rgba(112,116,128,${0.028 * sceneAlpha * (1 - mix)})`);
      gray.addColorStop(0.82, `rgba(164,168,179,${0.052 * sceneAlpha * (1 - mix)})`);
      gray.addColorStop(1, 'rgba(72,75,85,0)');
      ctx.fillStyle = gray;
      ctx.beginPath();
      ctx.arc(0, 0, floorRX * 1.10, 0, TAU);
      ctx.fill();
    }

    // 点亮后逐渐替换为参考图中的粉紫色粒子池。
    if (mix > 0.001) {
      const pink = ctx.createRadialGradient(0, 0, floorRX * 0.08, 0, 0, floorRX * 1.08);
      pink.addColorStop(0, `rgba(100,49,149,${0.015 * sceneAlpha * mix})`);
      pink.addColorStop(0.46, `rgba(147,63,218,${0.035 * sceneAlpha * mix})`);
      pink.addColorStop(0.80, `rgba(207,91,242,${0.075 * sceneAlpha * mix})`);
      pink.addColorStop(1, 'rgba(111,44,180,0)');
      ctx.fillStyle = pink;
      ctx.beginPath();
      ctx.arc(0, 0, floorRX * 1.10, 0, TAU);
      ctx.fill();
    }

    ctx.restore();
  }

  function drawCore(x, y, size, colorIndex, alpha, shape, pinkMix = 1) {
    const mix = clamp(pinkMix, 0, 1);
    const gray = GRAY_LEVELS[colorIndex];
    const pink = COLORS[colorIndex].rgb;
    const r = Math.round(lerp(gray, pink[0], mix));
    const g = Math.round(lerp(gray, pink[1], mix));
    const b = Math.round(lerp(gray, pink[2], mix));

    ctx.globalAlpha = alpha;
    ctx.fillStyle = `rgb(${r},${g},${b})`;

    if (shape < 0.38) {
      ctx.fillRect(x - size * 0.75, y - size * 0.35, size * 1.50, Math.max(0.75, size * 0.70));
    } else if (shape < 0.72) {
      ctx.fillRect(x - size * 0.34, y - size * 0.78, Math.max(0.72, size * 0.68), size * 1.56);
    } else {
      ctx.beginPath();
      ctx.arc(x, y, size * 0.62, 0, TAU);
      ctx.fill();
    }
  }

  function drawGlow(x, y, radius, alpha, pinkMix = 1) {
    const mix = clamp(pinkMix, 0, 1);

    if (grayGlowSprite && mix < 0.999) {
      ctx.globalAlpha = alpha * (1 - mix);
      ctx.drawImage(grayGlowSprite, x - radius, y - radius, radius * 2, radius * 2);
    }

    if (pinkGlowSprite && mix > 0.001) {
      ctx.globalAlpha = alpha * mix;
      ctx.drawImage(pinkGlowSprite, x - radius, y - radius, radius * 2, radius * 2);
    }
  }

  function drawFloor(time, deltaMs, pinkMix, sceneAlpha) {
    drawFloorGlow(pinkMix, sceneAlpha);
    ctx.globalCompositeOperation = 'lighter';

    // 后半圈先画，前半圈后画，椭圆会更有空间感。
    floorParticles.sort((a, b) => a.depth - b.depth);
    for (let i = 0; i < floorParticles.length; i++) {
      const p = floorParticles[i];
      p.update(time, deltaMs);
      const front = 0.72 + p.depth * 0.48;
      const size = p.size * front;
      const flicker = 0.82 + Math.sin(time * 0.0031 + p.phase) * 0.18;
      const alpha = p.alpha * flicker * (0.70 + p.depth * 0.38) * sceneAlpha;
      if (p.glow) drawGlow(p.x, p.y, size * 4.4, alpha * 0.17, pinkMix);
      drawCore(p.x, p.y, size, p.colorIndex, alpha, p.phase % 1, pinkMix);
    }

    ctx.globalAlpha = 1;
    ctx.globalCompositeOperation = 'source-over';
  }
